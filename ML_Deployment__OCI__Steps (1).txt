from sklearn.datasets import load_iris
from sklearn.ensemble import RandomForestClassifier
import joblib

X, y = load_iris(return_X_y=True)

'''
Loads the classic Iris dataset, which contains measurements of iris flowers and 
their species labels.
X = features (sepal length, sepal width, petal length, petal width)
y = target labels (species)

load_iris() is a utility function in scikit-learn that loads the classic Iris dataset.

By default, it returns a dictionary-like object called a Bunch that
 contains several fields (features, targets, metadata, etc.).

data = load_iris() -->dict 
>>> data.keys()   
dict_keys(['data', 'target', 'frame', 'target_names', 'DESCR', 'feature_names', 'filename'])

You then access your data like this:
X = data.data       # the feature matrix
y = data.target     # the target labels

So you need two lines to extract X and y.

X, y = load_iris(return_X_y=True)
It directly returns the features and targets as a tuple:
X: the feature matrix (numpy array)
y: the target vector (numpy array)
This is just a shortcut to avoid having to access .data and .target manually.
'''

# Create and train the model

model = RandomForestClassifier()
model.fit(X, y)
# Initializes a Random Forest classifier and trains it using the Iris data.


joblib.dump(model, "iris_model.pkl") # Save the trained model

print(" Model trained and saved.")

'''
Serializes (saves) the trained model to a file named iris_model.pkl.
You'll use this saved file when registering the model in Oracle Cloud Infrastructure (OCI).
'''
